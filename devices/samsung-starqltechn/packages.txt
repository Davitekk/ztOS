# ztOS on the Galaxy S9 (Snapdragon: SCV38, SM-G9600; starqltechn). The kernel
# is ztOS's own boot image, not a package (kernel/build-kernel.sh); the
# phone's firmware comes from its own partitions (ztos-starq-firmware).

# the boot stage's busybox (boot/build-boot.sh takes it from this root)
busybox

# the GPU's microcode (qcom/a630_sqe.fw, a630_gmu.bin); its zap shader is the
# phone's own
linux-firmware-qcom

# sound: ALSA's UCM, to which the layer adds this phone's profile
alsa-ucm-conf
# the speakers' limiter (fast_lookahead_limiter, in PipeWire's filter chain:
# overlay/etc/pipewire/pipewire.conf.d/60-ztos-starq-speakers.conf)
swh-plugins

# the modem over QRTR (ModemManager, qmicli for msm-modem-uim-selection)
modemmanager
libqmi
# where a call is heard (the UCM Voice Call verb, earpiece or speaker), for
# ztos-starq-mmbridge
callaudiod

# Bluetooth controller, management and scan tools
bluez
bluez-utils

# the sensors on the sensor DSP, through libssc
iio-sensor-proxy

# looking at the hardware
i2c-tools
pciutils
usbutils

# building the Qualcomm userspace and ModemManager in this root
# (userspace/build.sh; ModemManager's needs gdbus-codegen and xsltproc)
meson
glib2-devel
libxslt
